関数: mile2km
  │
  ├─定義
  │    ├─　目的：マイルの値をキロメートルに変換して返す
  │    ├─　引数
  │    │      └─ double mile :　マイルの値
  │    └─　戻り値
  │            └─ double : キロメートルの値
  │
  └─処理
      ├─マイルをキロメートルに変換して変数kmに代入する
      │      ├─ km = mile * 1.609
      │      └─ km を小数点以下2 桁までに四捨五入する
      │
      └─ km を返す

// 対応するＣ言語ソースコード
#include <math.h>
double mile2km(double mile)
{
    double km;
    km = mile * 1.609;
    /* 小数点以下2桁に四捨五入 */
    km = round(km * 100.0) / 100.0;
    return km;
}
