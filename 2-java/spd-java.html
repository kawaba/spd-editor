SPD for Java

Java言語用のSPDのパターンをまとめた。
Gemini Gemを使ってSPDからコードを生成した。
Gem用には「カスタム指示」に3400文字（約1000行）の指示を入力し、「知識」にpom.xmlを置いた。
全ての生成を行うための使用量は、無償ユーザーの一週間の上限の２％である。

1. コメント(※)

・※から行末まではコメントであり、SPDの構造・処理としては解釈しない。
・コメントは生成コードの対応する位置に // コメントとして転記する。
・引用符("…")・鉤括弧(「…」)内の※はコメント開始とみなさない。
・※だけの行は構造解析上は存在しないものとして扱う(縦線の連結を切らない)。

税込み価格の計算(※による1行コメント)
│
├─変数int priceにキーボードから価格を入力する ※税抜き価格
│  ※ここからは税込み価格の計算と表示
└─priceに1.1を掛けて整数に丸め、"税込=〇〇"の形式で表示する

import jp.kwebs.tools.Input;
public class Example1 {
    public static void main(String[] args) {
        int price = Input.getInt(); // 税抜き価格

        // ここからは税込み価格の計算と表示
        System.out.printf("税込=%d%n", Math.round(price * 1.1));
    }
}

import java.util.List;
import jp.kwebs.tools.Tools;
public class Example2_3_3 {
    public static void main(String[] args) {
        var list = List.of(10, 20, 30);
        int total = 0;

        for (var n : list) {
            total += n;
        }

        double average = (double) total / list.size();
        average = Tools.round(average, 1);

        System.out.printf("合計＝%d%n", total);
        System.out.printf("平均＝%.1f%n", average);
    }
}

import java.io.IOException;
import java.io.PrintWriter;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
public class Example2_3_8 {
    public static void main(String[] args) {
        var list = List.of(10, 20, 30);
        var path = Path.of("output.txt");

        try (var out = new PrintWriter(Files.newBufferedWriter(path))) {
            for (var line : list) {
                out.println(line);
            }
        } catch (IOException e) {
            System.out.println("ファイル出力エラー");
        }
    }
}

4.1 クラスメソッド

クラスメソッド：mile2km
│
├─目的：マイルを距離に変換する
├─引数
│    └─double mile：マイルの値
├─戻り値
│      └─double： キロメートルの値
│
└─処理
      ├─マイルをキロメートルに変換する
      │      ├─km = mile * 1.609
      │      └─km = km を小数点以下2 桁までに四捨五入する
      └─ km を返す

import jp.kwebs.tools.Tools;
public class Example4_1 {

    /**
     * マイルを距離に変換する
     * @param mile マイルの値
     * @return キロメートルの値
     */
    public static double mile2km(double mile) {
        double km = mile * 1.609;
        km = Tools.round(km, 2);
        return km;
    }
}

import java.util.Objects;
// 会員を表すクラス
public class Member {

    protected static final int FEE = 5000; // 会費額 (会議の基本月額)

    private int uid;     // 番号
    private String name; // 名前

    // 自動生成：コンストラクタ：引数なし
    public Member() {
    }

    // 自動生成：コンストラクタ（全引数）
    public Member(int uid, String name) {
        this.uid = uid;
        this.name = name;
    }

    // 自動生成：ゲッター
    public int getUid() {
        return uid;
    }

    public String getName() {
        return name;
    }

    // 自動生成：セッター
    public void setUid(int uid) {
        this.uid = uid;
    }

    public void setName(String name) {
        this.name = name;
    }

    // 自動生成：toString
    @Override
    public String toString() {
        return "Member[uid=" + uid + ", name=" + name + "]";
    }

    // 自動生成：等値判定 ← uid
    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (!(obj instanceof Member other)) {
            return false;
        }
        return uid == other.uid;
    }

    @Override
    public int hashCode() {
        return Objects.hash(uid);
    }

    /**
     * 会費額を返す
     * @return 会費の額
     */
    public int membership_fee() {
        return FEE;
    }

    /** クラスメソッドの例を示す */
    public static void doSomethig() {
        // 詳細を省略（何かの処理を書く）
    }

    public static void main(String[] args) {
        var member = new Member(100, "田中宏");
        System.out.println(member);
    }
}

  └─メソッド：add
        ├─目的：CSV 文字列に新しい文字列を連結する
        ├─引数
        │    └─追加する文字列：String s
        ├─戻り値：なし
        └─処理
              └─◇─csv は空文字である
                  │    └─csv にs を代入する
                  └─else
                        └─ csvに、delimiter + s を連結する

// CSV 文字列を生成する
public class Joining {

    private String csv;       // CSV 文字列
    private String delimiter; // 区切り文字

    // 自動生成：コンストラクタ（全引数）
    public Joining(String csv, String delimiter) {
        this.csv = csv;
        this.delimiter = delimiter;
    }

    // 引数が1 つのコンストラクタ
    public Joining(String delimiter) {
        this.csv = "";
        this.delimiter = delimiter;
    }

    // 引数のないコンストラクタ
    public Joining() {
        this.csv = "";
        this.delimiter = ",";
    }

    // 自動生成：ゲッター
    public String getCsv() {
        return csv;
    }

    public String getDelimiter() {
        return delimiter;
    }

    /**
     * CSV 文字列に新しい文字列を連結する
     * @param s 追加する文字列
     */
    public void add(String s) {
        if (csv.isEmpty()) {
            csv = s;
        } else {
            csv = csv + delimiter + s;
        }
    }
}

└─インスタンスメソッド：age
      │

      ├─引数：なし
      ├─戻り値：int 、年齢
      └─処理
            ├─today ← 今日の日付
            ├─period ← 誕生日(birthday)から今日(today )までの期間を計算する
            └─periodから年の値を取り出して返す

import java.time.LocalDate;
import java.time.Period;
// シニア会員を表すクラス
public class SeniorMember extends Member {

    private LocalDate birthday; // 誕生日

    // 自動生成：コンストラクタ（全引数）
    public SeniorMember(int uid, String name, LocalDate birthday) {
        super(uid, name);
        this.birthday = birthday;
    }

    // 自動生成：ゲッター
    public LocalDate getBirthday() {
        return birthday;
    }

    // 自動生成：toString
    @Override
    public String toString() {
        return "SeniorMember[uid=" + getUid() + ", name=" + getName() + ", birthday=" + birthday + "]";
    }

    /**
     * 年齢を返す
     * @return 年齢
     */
    public int age() {
        var today = LocalDate.now();
        var period = Period.between(birthday, today);
        return period.getYears();
    }
}
